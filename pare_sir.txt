************ Create System Socket File for Gunicorn *******

Example:-  nano /etc/systemd/system/crenta.in.gunicorn.socket

cd  /etc/systemd/system/

 nano pare_sir.com.gunicorn.socket



[Unit]
Description=pare_sir.com.gunicorn socket

[Socket]

[Install]
WantedBy=sockets.target


************** Create System Service File for Gunicorn ******
Syntax:-  nano /etc/systemd/system/your_domain.gunicorn.service
Example:-  nano /etc/systemd/system/pare_sir.com.gunicorn.service

cd  /etc/systemd/system/

 nano pare_sir.com.gunicorn.service


[Unit]
Description=pare_sir.com.gunicorn daemon
Requires=pare_sir.com.gunicorn.socket
After=network.target

[Service]
User=root
Group=root
WorkingDirectory=/root/pare_sir
ExecStart=/root/pare_sir/venv/bin/gunicorn \
          --access-logfile - \
          --workers 3 \
          --bind unix:/run/pare_sir.com.gunicorn.sock \
          pare_sir.wsgi:application

[Install]
WantedBy=multi-user.target

----------------------------------------------------------------
systemctl start pare_sir.com.gunicorn.socket

systemctl start pare_sir.com.gunicorn.service

-----------------------------------------------------------------

systemctl enable pare_sir.com.gunicorn.socket

systemctl enable pare_sir.com.gunicorn.service

-----------------------------------------------------------------------

systemctl status pare_sir.com.gunicorn.socket

systemctl status pare_sir.com.gunicorn.service

-------------------------------------------

systemctl daemon-reload
systemctl restart pare_sir.com.gunicorn

****************** Create Virtual Host File ******
Syntax:-  nano /etc/nginx/sites-available/your_domain
Example:-  nano /etc/nginx/sites-available/crenta.in


cd /etc/nginx/sites-available

 nano pare_sir.com



server{
    listen 80;
    listen [::]:80;

    server_name pare_sir.tejorder.com www.pare_sir.tejorder.com;
    client_max_body_size 500M;

    location = /favicon.ico { access_log off; log_not_found off; }

    location / {
        proxy_set_header Host $http_host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass http://unix:/run/pare_sir.com.gunicorn.sock;
    }

    
    
}
